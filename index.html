app.py Python 3.11+ 
from quantumgui import Theme, ui

theme = Theme(seed="#6750A4", mode="dark")
nombre = ui.state("")

with ui.window("Mi App", theme=theme) as win:
    with ui.column(gap=16).padding(24):
        ui.text("Bienvenido", role="headline_medium")

        ui.text_field(
            "Tu nombre",
            variant="outlined",
            leading_icon="person",
            bind=nombre,
        ).fill_width()

        ui.button(
            "Saludar",
            variant="filled",
            on_click=lambda: ui.snackbar(f"Hola {nombre.value}!"),
        )

win.run()
